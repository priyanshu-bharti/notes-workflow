topic: "Arrays - 1D"
problem: "Largest Element"
difficulty: "Easy"
description: "Given an array, we have to find the largest element in the array."
use-cases: "Finding out the largest item in the unsorted array in single loop."
time-complexity: "O(N) for optimal, O(N^2) for bruteforce"
space-complexity: "O(1)"
naive-approach: "Assume that the first item is the largest one, then loop through all the items in the array, then loop through the next list item in a nested loop then compare if the next item is equal or greater than the max item."
optimal-approach: "Assume that the max item is too small, then loop through the items in the list, and check if the curr item is greater or equal to the max value."
---

```ts
function bruteforce(arr: number[]) {
  // Assume that the first element is the largest one.
  let max = arr[0];
  // Loop through the element in the array.
  for (let i = 0; i < arr.length; i++) {
    // Loop through the next item in the array.
    for (let j = i + 1; j < arr.length; j++) {
      // If the next item is equal or greater, make it the max
      if (arr[j] >= max) {
        max = arr[j];
      }
    }
  }
  return max;
}

function optimal(arr: number[]) {
  // Assume the max element is too small.
  let max: number = -Infinity;
  // Loop through each item in the list
  for (let i of arr) {
    // If the current item is equal or greater than max, make it max.
    if (i >= max) {
      max = i;
    }
  }
  return max;
}

const inputs = [
  [3, 3, 6, 1],
  [3, 3, 0, 99, -40],
];

// Run the tests for bruteforce approach
for (let input of inputs) {
  const result = bruteforce(input);
  console.log(result);
}

// Run the tests for optimal approach
for (let input of inputs) {
  const result = optimal(input);
  console.log(result);
}
```
